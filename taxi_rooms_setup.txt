-- 2. 택시 합승 방 테이블 생성 (is_active 및 chat_url 포함)
create table if not exists taxi_rooms (
  id bigserial primary key,
  creator_id text not null,
  creator_name text not null,
  start_location_name text not null,
  start_lat float8 not null,
  start_lng float8 not null,
  end_location_name text not null,
  end_lat float8 not null,
  end_lng float8 not null,
  departure_time timestamptz not null,
  max_people int not null default 4,
  participants text[] default '{}',
  chat_url text, -- 카카오톡 오픈채팅 링크
  is_active boolean default true, -- 방 활성화 상태 (추가됨)
  created_at timestamptz default now()
);

-- 3. 실시간(Realtime) 업데이트 활성화
-- 이미 추가되어 있을 경우 에러가 날 수 있으니, 필요한 경우에만 실행하세요.

-- 4. RLS(Row Level Security) 활성화
alter table taxi_rooms enable row level security;

-- 5. 보안 정책(Policy) 설정
-- (이미 정책이 존재하면 에러가 날 수 있으니 확인 후 실행하세요)

-- 누구나 방 목록 보기
create policy "누구나 방 목록 보기" on taxi_rooms for select to public using (true);

-- 인증된 유저 방 생성
create policy "인증된 유저 방 생성" on taxi_rooms for insert to authenticated with check (auth.uid()::text = creator_id);

-- 인증된 유저 방 수정 (상태 변경 포함)
create policy "인증된 유저 방 수정" on taxi_rooms for update to authenticated using (true);

-- 방 삭제 정책: 방장이고 본인 혼자일 때만 삭제 허용
create policy "방장은 혼자일 때만 방 삭제 가능" 
on taxi_rooms for delete 
to authenticated 
using (
  (auth.uid())::text = creator_id 
  and 
  (participants is null or array_length(participants, 1) <= 1)
);

-- 6. 자동 삭제 설정 (pg_cron)
-- 만료된 방을 1분마다 체크하여 약속시간 5분이 지난 방 자동 정리
create extension if not exists pg_cron;
select cron.schedule(
  'delete-expired-rooms',
  '* * * * *', -- 매 분마다 실행
  $$ delete from taxi_rooms where departure_time < now() - interval '5 minutes' $$
);

-- 10. 방 삭제 시 채팅 링크 자동 반환 트리거 (필수)
-- 방이 삭제될 때(자동 삭제 포함) 해당 방이 사용하던 링크를 chat_pool에서 다시 사용 가능으로 변경합니다.
CREATE OR REPLACE FUNCTION public.handle_room_deletion()
RETURNS TRIGGER AS $$
BEGIN
  IF OLD.chat_url IS NOT NULL THEN
    UPDATE public.chat_pool
    SET is_used = false
    WHERE url = OLD.chat_url;
  END IF;
  RETURN OLD;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS tr_on_room_deleted ON public.taxi_rooms;
CREATE TRIGGER tr_on_room_deleted
AFTER DELETE ON public.taxi_rooms
FOR EACH ROW
EXECUTE FUNCTION public.handle_room_deletion();

-- 8. 채팅 링크 풀 테이블 생성
create table if not exists chat_pool (
  id bigserial primary key,
  url text not null unique,
  usage_count int default 0,
  is_used boolean default false, -- 현재 방에서 사용 중인지 여부 (추가됨)
  last_used_at timestamptz,
  created_at timestamptz default now()
);

-- 9. 기존 chat_pool 테이블에 is_used 컬럼 추가 (필수)
-- 앱에서 링크 사용 여부를 추적하기 위해 아래 명령어를 Supabase SQL Editor에서 실행하세요.
ALTER TABLE public.chat_pool ADD COLUMN IF NOT EXISTS is_used boolean DEFAULT false;
