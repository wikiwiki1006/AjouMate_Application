  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  name text NOT NULL,
  address text,
  lat double precision DEFAULT 0,
  lng double precision DEFAULT 0,
  sort_order int DEFAULT 0,
  created_at timestamp with time zone DEFAULT now()
);
