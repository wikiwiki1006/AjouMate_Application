ALTER TABLE public.frequent_locations ENABLE ROW LEVEL SECURITY;

-- 2. 누구나 데이터를 읽을 수 있도록 허용하는 정책(Policy) 생성
-- 기존 정책이 있다면 삭제하고 새로 만듭니다.
DROP POLICY IF EXISTS "Allow public read access" ON public.frequent_locations;

CREATE POLICY "Allow public read access"
ON public.frequent_locations FOR SELECT
TO public
USING (true);
